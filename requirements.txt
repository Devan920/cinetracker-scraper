name: CineTracker Live Scraper

on:
  schedule:
    - cron: '*/30 * * * *' # ഓരോ 30 മിനിറ്റിലും തനിയെ റൺ ആവും
  workflow_dispatch:       # GitHub-ൽ നിന്ന് നേരിട്ട് Run ക്ലിക്ക് ചെയ്യാനും പറ്റും

jobs:
  scrape:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Repo
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.11'

      - name: Install Dependencies
        run: pip install -r requirements.txt

      - name: Run Box Office Scraper
        env:
          SUPABASE_URL: "https://pblkvrybekogmweyajht.supabase.co"
          SUPABASE_SERVICE_KEY: ${{ secrets.SUPABASE_SERVICE_KEY }}
          TMDB_API_KEY: ${{ secrets.TMDB_API_KEY }}
        run: python scraper.py
